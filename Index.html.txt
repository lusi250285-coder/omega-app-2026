    <script>
        // تهيئة الخريطة على إحداثيات افتراضية (المغرب / العالم)
        const map = L.map('map').setView([31.7917, -7.0926], 6);

        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
            maxZoom: 19,
            attribution: '© OpenStreetMap contributors'
        }).addTo(map);

        const marker = L.marker([31.7917, -7.0926]).addTo(map)
            .bindPopup('النقطة المركزية النشطة')
            .openPopup();

        function searchLocation() {
            const query = document.getElementById('cityInput').value;
            if(!query) return;
            
            // استخدام خدمة Geocoding الحقيقية من Nominatim
            fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}`)
                .then(response => response.json())
                .then(data => {
                    if(data && data.length > 0) {
                        const lat = parseFloat(data[0].lat);
                        const lon = parseFloat(data[0].lon);
                        map.setView([lat, lon], 12);
                        marker.setLatLng([lat, lon]).setPopupContent(data[0].display_name).openPopup();
                        
                        // جلب الطقس الحقيقي عبر Open-Meteo
                        fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current_weather=true`)
                            .then(res => res.json())
                            .then(weather => {
                                if(weather && weather.current_weather) {
                                    document.getElementById('tempVal').innerText = weather.current_weather.temperature + " °C";
                                }
                            });
                    } else {
                        alert('لم يتم العثور على الموقع المطلوبة.');
                    }
                }).catch(err => console.error(err));
        }
    </script>
